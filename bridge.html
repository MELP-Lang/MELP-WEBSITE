-- bridge.mlp — CAN ↔ ETH Bridge (PDA_37 FAZ 2)
-- MELP scope/custody: tek satır manuel cleanup YOK

external function mlp_socket_create() as numeric
external function mlp_socket_connect(numeric fd; string addr; numeric port) as numeric
external function mlp_socket_send(numeric fd; string data) as numeric
external function mlp_socket_recv(numeric fd; numeric max) as string
external function println(string s) as numeric

numeric function main()
    println("=== MELP Bridge POC: CAN → ETH ===")

    -- CAN: baglan, mesaj al
    numeric can_fd = mlp_socket_create()
    mlp_socket_connect(can_fd; "127.0.0.1"; 9001)
    string can_msg = mlp_socket_recv(can_fd; 256)

    -- ETH: baglan, mesaji ilet
    numeric eth_fd = mlp_socket_create()
    mlp_socket_connect(eth_fd; "127.0.0.1"; 9002)
    mlp_socket_send(eth_fd; "BRIDGE_MSG: " & can_msg)

    println("✅ Bridge POC PASS")
    return 0
end function
